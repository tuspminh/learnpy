## 3\. Lớp Domain (Domain Layer)

Lớp này không phụ thuộc vào bất kỳ thư viện bên ngoài nào (ngoại trừ dataclasses chuẩn của Python).

## `domain/models.py`
    
    
    from dataclasses import dataclass, field
    from pathlib import Path
    from typing import List
    
    @dataclass(frozen=True)
    def TTSConfig:
        engine: str
        voice: str | None
        rate: str | None
        volume: str | None
        pitch: str | None
        chunk_size: int = 2000
        merge_audio: bool = True
    
        def __post_init__(self):
            if not (1500 <= self.chunk_size <= 2500):
                raise ValueError("Chunk size phải nằm trong khoảng từ 1500 đến 2500 ký tự.")
    
    @dataclass
    class TextChunk:
        index: int
        text: str
        language: str
    
    @dataclass
    class TextFileDocument:
        file_path: Path
        content: str
        language: str
        chunks: List[TextChunk] = field(default_factory=list)
    
    @dataclass
    class AudioArtifact:
        file_path: Path
        duration_seconds: float = 0.0
    

## `domain/interfaces.py`
    
    
    from abc import ABC, abstractmethod
    from pathlib import Path
    from typing import List
    from .models import TextChunk, TTSConfig, AudioArtifact
    
    class ITTSEngine(ABC):
        @abstractmethod
        def synthesize(self, chunk: TextChunk, config: TTSConfig, output_path: Path) -> AudioArtifact:
            pass
    
    class IAudioMerger(ABC):
        @abstractmethod
        def merge(self, audio_paths: List[Path], output_path: Path) -> AudioArtifact:
            pass
    
    class ITextProcessor(ABC):
        @abstractmethod
        def split_into_sentences(self, text: str) -> List[str]:
            pass
    

## `domain/services.py`
    
    
    from typing import List
    from .models import TextChunk
    from .interfaces import ITextProcessor
    
    class TextChunkerService:
        def __init__(self, text_processor: ITextProcessor):
            self._processor = text_processor
    
        def chunk_text(self, text: str, lang: str, min_size: int = 1500, target_size: int = 2000, max_size: int = 2500) -> List[TextChunk]:
            sentences = self._processor.split_into_sentences(text)
            chunks: List[TextChunk] = []
            current_chunk = []
            current_length = 0
            chunk_index = 1
    
            for sentence in sentences:
                sentence_len = len(sentence)
                
                if current_length + sentence_len > max_size and current_length >= min_size:
                    # Đóng chunk hiện tại khi vượt ngưỡng tối đa
                    combined_text = " ".join(current_chunk)
                    chunks.append(TextChunk(index=chunk_index, text=combined_text, language=lang))
                    chunk_index += 1
                    current_chunk = [sentence]
                    current_length = sentence_len
                else:
                    current_chunk.append(sentence)
                    current_length += sentence_len
    
            # Xử lý phần văn bản còn lại cuối cùng
            if current_chunk:
                combined_text = " ".join(current_chunk)
                chunks.append(TextChunk(index=chunk_index, text=combined_text, language=lang))
    
            return chunks

## `infrastructure/text_processing/underthesea_processor.py`
    
    
    from typing import List
    from tts_cli.domain.interfaces import ITextProcessor
    from underthesea import sent_tokenize
    
    class VietnameseTextProcessor(ITextProcessor):
        def split_into_sentences(self, text: str) -> List[str]:
            if not text.strip():
                return []
            return sent_tokenize(text)
    

## `infrastructure/text_processing/spacy_processor.py`
    
    
    from typing import List
    import spacy
    from tts_cli.domain.interfaces import ITextProcessor
    
    class EnglishTextProcessor(ITextProcessor):
        def __init__(self):
            # Tải mô hình nhỏ của spacy cho tiếng Anh
            try:
                self.nlp = spacy.load("en_core_web_sm")
            except OSError:
                from spacy.cli import download
                download("en_core_web_sm")
                self.nlp = spacy.load("en_core_web_sm")
    
        def split_into_sentences(self, text: str) -> List[str]:
            if not text.strip():
                return []
            doc = self.nlp(text)
            return [sent.text.strip() for sent in doc.sents]
    

## `infrastructure/tts_engines/edge_engine.py`
    
    
    import asyncio
    from pathlib import Path
    import edge_tts
    from tts_cli.domain.interfaces import ITTSEngine
    from tts_cli.domain.models import TextChunk, TTSConfig, AudioArtifact
    
    class EdgeTTSEngine(ITTSEngine):
        def synthesize(self, chunk: TextChunk, config: TTSConfig, output_path: Path) -> AudioArtifact:
            # Chọn voice mặc định dựa trên ngôn ngữ nếu cli không truyền vào
            voice = config.voice or ("vi-VN-HoaiAnNeural" if chunk.language == "vi" else "en-US-AriaNeural")
            
            # Thiết lập các tham số bổ sung dựa theo định dạng của edge-tts (ví dụ: '+10%', '-10%')
            rate = config.rate or "+0%"
            volume = config.volume or "+0%"
            pitch = config.pitch or "+0Hz"
    
            async def _run():
                communicate = edge_tts.Communicate(
                    text=chunk.text, 
                    voice=voice, 
                    rate=rate, 
                    volume=volume, 
                    pitch=pitch
                )
                await communicate.save(str(output_path))
    
            asyncio.run(_run())
            return AudioArtifact(file_path=output_path)
    

## `infrastructure/audio/pydub_merger.py`
    
    
    from typing import List
    from pathlib import Path
    from pydub import AudioSegment
    from tts_cli.domain.interfaces import IAudioMerger
    from tts_cli.domain.models import AudioArtifact
    
    class PydubAudioMerger(IAudioMerger):
        def merge(self, audio_paths: List[Path], output_path: Path) -> AudioArtifact:
            if not audio_paths:
                raise ValueError("Không có file âm thanh nào để gộp.")
            
            # Khởi tạo segment đầu tiên
            combined = AudioSegment.from_file(audio_paths[0])
            
            for path in audio_paths[1:]:
                next_segment = AudioSegment.from_file(path)
                combined += next_segment
                
            combined.export(output_path, format="mp3")
            return AudioArtifact(file_path=output_path, duration_seconds=combined.duration_seconds)

## 5\. Lớp Use Cases (Application Layer)

Điều phối dòng chảy nghiệp vụ từ việc đọc file, phân tách văn bản, gọi bộ chuyển đổi giọng nói đến gộp tệp.

## `use_cases/process_tts.py`
    
    
    from pathlib import Path
    from typing import Dict
    from tts_cli.domain.models import TTSConfig, TextFileDocument
    from tts_cli.domain.interfaces import ITTSEngine, IAudioMerger, ITextProcessor
    from tts_cli.domain.services import TextChunkerService
    
    class ProcessTextToSpeechUseCase:
        def __init__(
            self, 
            processors: Dict[str, ITextProcessor], 
            tts_engines: Dict[str, ITTSEngine], 
            audio_merger: IAudioMerger
        ):
            self._processors = processors
            self._tts_engines = tts_engines
            self._audio_merger = audio_merger
    
        def execute(self, target_path: Path, config: TTSConfig, detected_lang: str):
            # 1. Xác định file cần xử lý (Hỗ trợ cả file đơn lẻ lẫn thư mục)
            files_to_process = [target_path] if target_path.is_file() else list(target_path.glob("*.txt"))
    
            for file_path in files_to_process:
                with open(file_path, "r", encoding="utf-8") as f:
                    content = f.read()
    
                # 2. Lấy bộ xử lý ngôn ngữ tương ứng (OCP trong SOLID)
                processor = self._processors.get(detected_lang)
                if not processor:
                    raise ValueError(f"Ngôn ngữ {detected_lang} không được hỗ trợ xử lý ngữ cảnh.")
    
                chunker_service = TextChunkerService(processor)
                chunks = chunker_service.chunk_text(
                    text=content, 
                    lang=detected_lang, 
                    target_size=config.chunk_size
                )
    
                # 3. Tạo thư mục output riêng cho từng file để lưu chunk audio
                output_dir = file_path.parent / file_path.stem
                output_dir.mkdir(parents=True, exist_ok=True)
    
                # 4. Lấy TTS Engine cấu hình bởi người dùng
                engine = self._tts_engines.get(config.engine)
                if not engine:
                    raise ValueError(f"Engine {config.engine} chưa được cài đặt hệ thống.")
    
                chunk_audio_paths = []
                for chunk in chunks:
                    chunk_file_path = output_dir / f"chunk_{chunk.index:03d}.mp3"
                    engine.synthesize(chunk, config, chunk_file_path)
                    chunk_audio_paths.append(chunk_file_path)
    
                # 5. Merge âm thanh nếu không bật cờ --no-merge
                if config.merge_audio and chunk_audio_paths:
                    merged_output_path = file_path.parent / f"{file_path.stem}_merged.mp3"
                    self._audio_merger.merge(chunk_audio_paths, merged_output_path)

## 6\. Giao diện CLI (Presentation / Interface Layer)

Sử dụng thư viện `click` để tiếp nhận dữ liệu từ dòng lệnh và gọi Use Case.

## `main.py`
    
    
    import click
    from pathlib import Path
    from tts_cli.domain.models import TTSConfig
    from tts_cli.use_cases.process_tts import ProcessTextToSpeechUseCase
    from tts_cli.infrastructure.text_processing.underthesea_processor import VietnameseTextProcessor
    from tts_cli.infrastructure.text_processing.spacy_processor import EnglishTextProcessor
    from tts_cli.infrastructure.tts_engines.edge_engine import EdgeTTSEngine
    from tts_cli.infrastructure.audio.pydub_merger import PydubAudioMerger
    
    @click.command()
    @click.argument("input-path", type=click.Path(exists=True, path_type=Path))
    @click.option("--lang", type=click.Choice(["vi", "en"]), default="vi", help="Ngôn ngữ xử lý văn bản")
    @click.option("--tts-engine", type=click.Choice(["system", "edge", "google-translate", "google-cloud"]), default="edge")
    @click.option("--voice", type=str, default=None)
    @click.option("--rate", type=str, default=None)
    @click.option("--volume", type=str, default=None)
    @click.option("--pitch", type=str, default=None)
    @click.option("--no-merge", is_flag=True, default=False, help="Không gộp các file audio chunk thành một file")
    @click.option("--chunk-size", type=int, default=2000, help="Kích thước ký tự tối ưu cho một chunk (1500 - 2500)")
    def cli(input_path, lang, tts_engine, voice, rate, volume, pitch, no_merge, chunk_size):
        """Ứng dụng CLI chuyển đổi file văn bản/thư mục thành giọng nói tiếng Anh hoặc tiếng Việt."""
        try:
            # Khởi tạo Configuration từ tham số nhận vào
            config = TTSConfig(
                engine=tts_engine,
                voice=voice,
                rate=rate,
                volume=volume,
                pitch=pitch,
                chunk_size=chunk_size,
                merge_audio=not no_merge
            )
    
            # Đăng ký các thành phần Dependency Injection thủ công (Đảm bảo DIP độc lập nguyên lý SOLID)
            processors = {
                "vi": VietnameseTextProcessor(),
                "en": EnglishTextProcessor()
            }
            
            tts_engines = {
                "edge": EdgeTTSEngine()
                # Có thể bổ sung thêm SystemTTSEngine(), GoogleTTSEngine() tại đây mà không ảnh hưởng mã nguồn cũ
            }
            
            merger = PydubAudioMerger()
    
            # Thực thi Use Case ứng dụng
            use_case = ProcessTextToSpeechUseCase(processors, tts_engines, merger)
            use_case.execute(input_path, config, lang)
            
            click.echo(click.style("Xử lý Text-to-Speech thành công hoàn tất!", fg="green"))
    
        except Exception as e:
            click.echo(click.style(f"Lỗi hệ thống: {str(e)}", fg="red"), err=True)
    
    if __name__ == "__main__":
        cli()

## 7\. Nguyên lý SOLID được thể hiện như thế nào trong Code?

  1. Single Responsibility Principle (SRP): `TextChunkerService` chỉ có một nhiệm vụ duy nhất là bóc tách chữ thành các đoạn có nghĩa, trong khi `EdgeTTSEngine` chịu trách nhiệm duy nhất là sinh ra file âm thanh từ chữ. Giao diện Click (`main.py`) chỉ lo thu thập tham số đầu vào.
  2. Open/Closed Principle (OCP): Khi bạn cần mở rộng một TTS Engine mới (ví dụ: `GoogleCloudTTSEngine`), bạn chỉ cần kế thừa lại `ITTSEngine` và đăng ký nó ở tầng CLI mà không phải sửa đổi bất kì dòng code xử lý lõi nào ở `ProcessTextToSpeechUseCase`.
  3. Liskov Substitution Principle (LSP): Tất cả các bộ xử lý ngôn ngữ (`VietnameseTextProcessor`, `EnglishTextProcessor`) đều tuân thủ chặt chẽ hành vi của interface `ITextProcessor` mà không làm thay đổi tính đúng đắn của chương trình khi hoán đổi.
  4. Interface Segregation Principle (ISP): Các interface được bóc tách nhỏ gọn: `ITTSEngine`, `IAudioMerger`, `ITextProcessor`. Một Client sử dụng bộ chuyển đổi âm thanh không bắt buộc phải kế thừa phương thức xử lý chuỗi văn bản.
  5. Dependency Inversion Principle (DIP): Lớp nghiệp vụ cao cấp `ProcessTextToSpeechUseCase` không phụ thuộc trực tiếp vào tầng công nghệ như `spacy`, `underthesea`, hay `edge_tts`. Nó phụ thuộc hoàn toàn vào các Interface trừu tượng trong thư mục `domain/interfaces.py`.



Để tiếp tục phát triển ứng dụng này, bạn có muốn tôi viết chi tiết phần hiện thực cho Google Cloud TTS Engine hay bộ cài đặt System TTS Engine cục bộ không?
