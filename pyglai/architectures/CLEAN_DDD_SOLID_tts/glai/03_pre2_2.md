## 🏗️ Kiến trúc Hệ thống & Luồng Dữ liệu (DIP)

Hệ thống tuân thủ nguyên lý đảo ngược phụ thuộc (Dependency Inversion - DIP). Lớp lõi (Domain) quy định luật chơi, lớp ứng dụng (Use Cases) điều phối luồng, lớp ngoài cùng (Infrastructure / Presentation) chỉ là chi tiết cài đặt dễ dàng thay thế.
    
    
    [ Presentation: Click CLI ] ──> [ Use Cases: ConvertTextToSpeech ] 
                                          │
                                          ▼
                                    [ Domain Layer ]
                         (Interfaces / Entities / Domain Services)
                                          ▲
                                          │
              [ Infrastructure: NLP (SpaCy/Underthesea) | TTS Engines | FFmpeg ]

## 💻 Mã nguồn Chi tiết các Lớp

## 1\. Domain Layer (`domain/`)

Thành phần cốt lõi, không chứa bất kỳ mã nguồn framework hay thư viện ngoài nào (Thỏa mãn Single Responsibility Principle - SRP).
    
    
    # domain/models.py
    from dataclasses import dataclass
    from typing import Optional, NamedTuple
    
    class Language:
        VI = "vi"
        EN = "en"
    
    @dataclass(frozen=True)
    class TTSConfig:
        engine_name: str
        voice: str
        rate: int
        volume: float
        pitch: Optional[float]
        chunk_size: int = 2000
    
    class TextChunk(NamedTuple):
        content: str
        sequence_number: int
        language: str
    
    @dataclass(frozen=True)
    class AudioFile:
        file_path: str
    
    
    
    # domain/interfaces.py
    from abc import ABC, abstractmethod
    from typing import List
    from domain.models import TextChunk, TTSConfig, AudioFile
    
    class ISentenceTokenizer(ABC):
        @abstractmethod
        def tokenize(self, text: str) -> List[str]:
            pass
    
    class ITTSEngine(ABC):
        @abstractmethod
        def synthesize(self, chunk: TextChunk, config: TTSConfig, output_path: str) -> AudioFile:
            pass
    
    class IAudioMerger(ABC):
        @abstractmethod
        def merge(self, audio_files: List[AudioFile], output_path: str) -> AudioFile:
            pass
    
    class IFileSystem(ABC):
        @abstractmethod
        def scan_files(self, path: str) -> List[str]: pass
        @abstractmethod
        def read_text(self, file_path: str) -> str: pass
        @abstractmethod
        def delete_file(self, file_path: str) -> None: pass
    

Thuật toán tách câu giữ ngữ cảnh nằm trọn vẹn trong Domain Service, cam kết giới hạn độ dài `1500 <= size <= 2500` ký tự.
    
    
    # domain/services.py
    from typing import List
    from domain.interfaces import ISentenceTokenizer
    from domain.models import TextChunk
    
    class TextChunkerService:
        def __init__(self, tokenizer: ISentenceTokenizer, language: str):
            self._tokenizer = tokenizer
            self._language = language
    
        def chunk_text(self, text: str, default_size: int = 2000) -> List[TextChunk]:
            sentences = self._tokenizer.tokenize(text)
            chunks: List[TextChunk] = []
            current_sentences: List[str] = []
            current_len = 0
            seq = 0
    
            for sentence in sentences:
                sentence = sentence.strip()
                if not sentence:
                    continue
                
                sentence_len = len(sentence)
                # Kiểm tra vượt ngưỡng Max 2500 hoặc đạt ngưỡng Default/Min mong muốn
                if current_len + sentence_len > 2500 or (current_len >= default_size and current_len + sentence_len > default_size):
                    if current_len >= 1500:
                        chunks.append(TextChunk(" ".join(current_sentences), seq, self._language))
                        current_sentences = [sentence]
                        current_len = sentence_len
                        seq += 1
                        continue
                
                current_sentences.append(sentence)
                current_len += sentence_len + 1 # Cộng 1 ký tự khoảng trắng
    
            if current_sentences:
                chunks.append(TextChunk(" ".join(current_sentences), seq, self._language))
                
            return chunks
    

## 2\. Infrastructure Layer (`infrastructure/`)

Nơi hiện thực hóa tất cả các Interface trừu tượng của Domain dựa trên các thư viện cụ thể.
    
    
    # infrastructure/nlp_splitters.py
    from domain.interfaces import ISentenceTokenizer
    from underthesea import sent_tokenize
    import spacy
    
    class VietnameseTokenizer(ISentenceTokenizer):
        def tokenize(self, text: str) -> list[str]:
            return sent_tokenize(text)
    
    class EnglishTokenizer(ISentenceTokenizer):
        def __init__(self):
            # Sử dụng mô hình nhỏ của SpaCy tối ưu tốc độ
            self.nlp = spacy.load("en_core_web_sm", disable=["ner", "textcat", "lemmatizer"])
            
        def tokenize(self, text: str) -> list[str]:
            doc = self.nlp(text)
            return [sent.text for sent in doc.sents]
    

Áp dụng Open-Closed Principle (OCP): Thêm mới Engine bằng cách kế thừa `ITTSEngine` mà không tác động tới hệ thống cũ.
    
    
    # infrastructure/tts_engines.py
    import asyncio
    from domain.interfaces import ITTSEngine
    from domain.models import TextChunk, TTSConfig, AudioFile
    import edge_tts
    from gtts import gTTS
    import pyttsx3
    from google.cloud import texttospeech
    
    class EdgeTTSEngine(ITTSEngine):
        def synthesize(self, chunk: TextChunk, config: TTSConfig, output_path: str) -> AudioFile:
            async def _speak():
                # Đồng bộ định dạng rate/volume theo chuẩn API edge-tts
                rate_str = f"{config.rate:+}%" if config.rate >= 0 else f"{config.rate}%"
                communicate = edge_tts.Communicate(
                    text=chunk.content, voice=config.voice, rate=rate_str
                )
                await communicate.save(output_path)
            asyncio.run(_speak())
            return AudioFile(output_path)
    
    class GoogleTranslateEngine(ITTSEngine):
        def synthesize(self, chunk: TextChunk, config: TTSConfig, output_path: str) -> AudioFile:
            tts = gTTS(text=chunk.content, lang=chunk.language)
            tts.save(output_path)
            return AudioFile(output_path)
    
    class SystemTTSEngine(ITTSEngine):
        def synthesize(self, chunk: TextChunk, config: TTSConfig, output_path: str) -> AudioFile:
            engine = pyttsx3.init()
            engine.setProperty('rate', 150 + config.rate)
            engine.setProperty('volume', config.volume / 100.0)
            engine.save_to_file(chunk.content, output_path)
            engine.runAndWait()
            return AudioFile(output_path)
    
    class GoogleCloudTTSEngine(ITTSEngine):
        def synthesize(self, chunk: TextChunk, config: TTSConfig, output_path: str) -> AudioFile:
            client = texttospeech.TextToSpeechClient()
            s_input = texttospeech.SynthesisInput(text=chunk.content)
            voice = texttospeech.VoiceSelectionParams(
                language_code="vi-VN" if chunk.language == "vi" else "en-US", 
                name=config.voice
            )
            audio_config = texttospeech.AudioConfig(
                audio_encoding=texttospeech.AudioEncoding.MP3,
                speaking_rate=1.0 + (config.rate / 100.0),
                volume_gain_db=0.0 # Tùy chỉnh theo tài liệu Google API
            )
            response = client.synthesize_speech(input=s_input, voice=voice, audio_config=audio_config)
            with open(output_path, "wb") as out:
                out.write(response.audio_content)
            return AudioFile(output_path)
    
    
    
    # infrastructure/file_audio_services.py
    import os
    from typing import List
    from domain.interfaces import IAudioMerger, IFileSystem
    from domain.models import AudioFile
    from pydub import AudioSegment
    
    class LocalFileSystem(IFileSystem):
        def scan_files(self, path: str) -> List[str]:
            if os.path.isfile(path) and path.endswith('.txt'):
                return [path]
            elif os.path.isdir(path):
                return [os.path.join(path, f) for f in os.listdir(path) if f.endswith('.txt')]
            return []
    
        def read_text(self, file_path: str) -> str:
            with open(file_path, 'r', encoding='utf-8') as f:
                return f.read()
    
        def delete_file(self, file_path: str) -> None:
            if os.path.exists(file_path):
                os.remove(file_path)
    
    class PydubAudioMerger(IAudioMerger):
        def merge(self, audio_files: List[AudioFile], output_path: str) -> AudioFile:
            combined = AudioSegment.empty()
            for audio in audio_files:
                segment = AudioSegment.from_file(audio.file_path)
                combined += segment
            combined.export(output_path, format="mp3")
            return AudioFile(output_path)
    

## 3\. Use Cases Layer (`use_cases/`)

Lớp điều phối quy trình xử lý, nhận diện dữ liệu đầu vào độc lập hoàn toàn với CLI framework.
    
    
    # use_cases/convert_tts.py
    from typing import Dict
    from domain.interfaces import ITTSEngine, ISentenceTokenizer, IAudioMerger, IFileSystem
    from domain.models import TTSConfig
    from domain.services import TextChunkerService
    
    class ConvertTextToSpeechUseCase:
        def __init__(
            self,
            file_system: IFileSystem,
            tokenizers: Dict[str, ISentenceTokenizer],
            engines: Dict[str, ITTSEngine],
            audio_merger: IAudioMerger
        ):
            self._fs = file_system
            self._tokenizers = tokenizers
            self._engines = engines
            self._merger = audio_merger
    
        def execute(self, input_path: str, config: TTSConfig, merge: bool, language: str) -> None:
            target_files = self._fs.scan_files(input_path)
            if not target_files:
                raise ValueError(f"Không tìm thấy file hợp lệ tại: {input_path}")
    
            tokenizer = self._tokenizers.get(language)
            engine = self._engines.get(config.engine_name)
            if not tokenizer or not engine:
                raise ValueError("Cấu hình Ngôn ngữ hoặc Engine không chính xác.")
    
            chunker = TextChunkerService(tokenizer, language)
    
            for file_path in target_files:
                content = self._fs.read_text(file_path)
                chunks = chunker.chunk_text(content, config.chunk_size)
                
                temp_audio_files = []
                for chunk in chunks:
                    chunk_output = f"{file_path}_chunk_{chunk.sequence_number}.mp3"
                    audio_file = engine.synthesize(chunk, config, chunk_output)
                    temp_audio_files.append(audio_file)
    
                if merge and len(temp_audio_files) > 1:
                    final_output = f"{file_path.removesuffix('.txt')}_final.mp3"
                    self._merger.merge(temp_audio_files, final_output)
                    # Cleanup các chunk tạm sau khi merge xong
                    for file in temp_audio_files:
                        self._fs.delete_file(file.file_path)
    

## 4\. Presentation Layer (`presentation/`) & `main.py`

Nơi tiếp nhận các tham số dòng lệnh từ người dùng và đóng vai trò Composition Root để lắp ghép linh kiện.
    
    
    # presentation/cli.py
    import click
    from domain.models import TTSConfig
    from use_cases.convert_tts import ConvertTextToSpeechUseCase
    
    def build_cli(use_case: ConvertTextToSpeechUseCase):
        @click.command()
        @click.argument('path', type=click.Path(exists=True))
        @click.option('--engine', 'tts_engine', default='edge', type=click.Choice(['system', 'edge', 'google-translate', 'google-cloud']))
        @click.option('--voice', default='vi-VN-HoaiAnNeural')
        @click.option('--rate', default=0, type=int)
        @click.option('--volume', default=100.0, type=float)
        @click.option('--pitch', default=None, type=float)
        @click.option('--chunk-size', default=2000, type=click.IntRange(1500, 2500))
        @click.option('--merge', is_flag=True, help="Gộp các file chunk audio thành một file duy nhất")
        @click.option('--lang', default='vi', type=click.Choice(['vi', 'en']))
        def main_cli(path, tts_engine, voice, rate, volume, pitch, chunk_size, merge, lang):
            """Hệ thống CLI chuyển đổi File/Folder văn bản thành giọng nói tiếng Anh/Việt."""
            config = TTSConfig(
                engine_name=tts_engine, voice=voice, rate=rate,
                volume=volume, pitch=pitch, chunk_size=chunk_size
            )
            try:
                click.echo(f"🚀 Bắt đầu xử lý {path} bằng Engine [{tts_engine}]...")
                use_case.execute(path, config, merge, lang)
                click.secho("🎉 Chuyển đổi hoàn tất thành công!", fg="green")
            except Exception as e:
                click.secho(f"❌ Thất bại: {str(e)}", fg="red", err=True)
    
        return main_cli
    
    
    
    # main.py
    from infrastructure.file_audio_services import LocalFileSystem, PydubAudioMerger
    from infrastructure.nlp_splitters import VietnameseTokenizer, EnglishTokenizer
    from infrastructure.tts_engines import EdgeTTSEngine, GoogleTranslateEngine, SystemTTSEngine, GoogleCloudTTSEngine
    from use_cases.convert_tts import ConvertTextToSpeechUseCase
    from presentation.cli import build_cli
    
    def bootstrap():
        # 1. Khởi tạo cơ sở hạ tầng (Infrastructure)
        file_system = LocalFileSystem()
        audio_merger = PydubAudioMerger()
    
        tokenizers = {
            "vi": VietnameseTokenizer(),
            "en": EnglishTokenizer()
        }
    
        engines = {
            "edge": EdgeTTSEngine(),
            "google-translate": GoogleTranslateEngine(),
            "system": SystemTTSEngine(),
            "google-cloud": GoogleCloudTTSEngine()
        }
    
        # 2. Tiêm phụ thuộc (Dependency Injection) vào Use Case
        use_case = ConvertTextToSpeechUseCase(
            file_system=file_system,
            tokenizers=tokenizers,
            engines=engines,
            audio_merger=audio_merger
        )
    
        # 3. Kích hoạt giao diện ứng dụng (Presentation)
        cli = build_cli(use_case)
        cli()
    
    if __name__ == '__main__':
        bootstrap()
