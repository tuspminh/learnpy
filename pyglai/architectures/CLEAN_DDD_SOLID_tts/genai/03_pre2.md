    > Bạn là lập trình viên python, thiét kế chi tiếp app sau theo kiến trúc clean, domain driven design và nguyên lý solid:
    app chuyển text thành voice, tiếng việt hoặc tiếng anh. đầu vào là file text hoặc folder chứa các file text.
    app cli dùng click để nhận tham số dòng lệnh. có các tham số sau : file/ folder, tts-engine, voice, rate, volume, pitch, merge (nếu có thì merge các file audio của các chunk của file text thành một file audio), chunk-size(mặc định 2000 ký tự, max 2500)
    xử lý file text, dùng underthesea với tiếng việt và spacy với tiếng anh, tách file text thành các chunk min 1500 ký tự, default 2000, max 2500), tách theo câu, cố gắng giữ ngữ cảnh.
    tts-engine: tuỳ chọn engine hệ thống, edge, google translate, google cloud tts.
    python 3.15 với uv



Kiến trúc hệ thống chuẩn **Clean Architecture** kết hợp **Domain-Driven Design (DDD)** và **SOLID** cho ứng dụng CLI Text-to-Speech (TTS).

#### 1. Domain Layer (`src/domain/`)

**`value_objects.py`**

```python
from dataclasses import dataclass
from enum import StrEnum

class Language(StrEnum):
    VIETNAMESE = "vi"
    ENGLISH = "en"

class TTSEngineType(StrEnum):
    SYSTEM = "system"
    EDGE = "edge"
    GTRANSLATE = "gtranslate"
    GCLOUD = "gcloud"

@dataclass(frozen=True)
class VoiceSettings:
    voice: str | None = None
    rate: float = 1.0
    volume: float = 1.0
    pitch: float = 0.0

@dataclass(frozen=True)
class ChunkConfig:
    min_size: int = 1500
    default_size: int = 2000
    max_size: int = 2500

    def __post_init__(self) -> None:
        if not (self.min_size <= self.default_size <= self.max_size):
            raise ValueError("Kích thước chunk không hợp lệ: min <= default <= max")

```

**`ports.py`** (Interface Segregation Principle - ISP)

```python
from pathlib import Path
from typing import Protocol
from src.domain.entities import TextChunk, AudioChunk
from src.domain.value_objects import VoiceSettings, Language

class TextSegmenterProtocol(Protocol):
    def split_sentences(self, text: str) -> list[str]: ...

class TTSEngineProtocol(Protocol):
    async def synthesize(self, chunk: TextChunk, settings: VoiceSettings, output_path: Path) -> AudioChunk: ...

class AudioMergerProtocol(Protocol):
    def merge(self, audio_files: list[Path], output_path: Path) -> Path: ...

```

**`services.py`** (Text Chunking Strategy - Giữ ngữ cảnh câu)

```python
from src.domain.entities import TextChunk
from src.domain.value_objects import ChunkConfig, Language
from src.domain.ports import TextSegmenterProtocol

class TextSplittingDomainService:
    def __init__(self, segmenter: TextSegmenterProtocol, config: ChunkConfig):
        self._segmenter = segmenter
        self._config = config

    def create_chunks(self, text: str, lang: Language) -> list[TextChunk]:
        sentences = self._segmenter.split_sentences(text)
        chunks: list[TextChunk] = []
        current_sentences: list[str] = []
        current_length = 0
        chunk_idx = 0

        for sentence in sentences:
            sentence_len = len(sentence)
            
            # Quá kích thước max -> đóng chunk hiện tại
            if current_length + sentence_len > self._config.max_size and current_length >= self._config.min_size:
                chunks.append(TextChunk(index=chunk_idx, text=" ".join(current_sentences), language=lang.value))
                chunk_idx += 1
                current_sentences = []
                current_length = 0

            current_sentences.append(sentence)
            current_length += sentence_len

        if current_sentences:
            chunks.append(TextChunk(index=chunk_idx, text=" ".join(current_sentences), language=lang.value))

        return chunks

```

#### 2. Infrastructure Layer (`src/infrastructure/`)

**`segmenters/nlp_segmenters.py`**

```python
import spacy
from underthesea import sent_tokenize
from src.domain.ports import TextSegmenterProtocol

class VietnameseSegmenter(TextSegmenterProtocol):
    def split_sentences(self, text: str) -> list[str]:
        return sent_tokenize(text)

class EnglishSegmenter(TextSegmenterProtocol):
    def __init__(self) -> None:
        self._nlp = spacy.load("en_core_web_sm", disable=["ner", "tagger", "lemmatizer"])
        self._nlp.add_pipe("sentencizer")

    def split_sentences(self, text: str) -> list[str]:
        doc = self._nlp(text)
        return [sent.text.strip() for sent in doc.sents]

```

**`tts_engines/edge_engine.py`** (Ví dụ triển khai EdgeTTS Adapter)

```python
from pathlib import Path
import edge_tts
from src.domain.entities import TextChunk, AudioChunk
from src.domain.value_objects import VoiceSettings
from src.domain.ports import TTSEngineProtocol

class EdgeTTSEngine(TTSEngineProtocol):
    async def synthesize(self, chunk: TextChunk, settings: VoiceSettings, output_path: Path) -> AudioChunk:
        voice = settings.voice or ("vi-VN-HoaiMyNeural" if chunk.language == "vi" else "en-US-ChristopherNeural")
        
        # Format rate/volume/pitch cho EdgeTTS
        rate_str = f"{int((settings.rate - 1) * 100):+d}%"
        volume_str = f"{int((settings.volume - 1) * 100):+d}%"
        pitch_str = f"{int(settings.pitch):+d}Hz"

        communicate = edge_tts.Communicate(
            text=chunk.text,
            voice=voice,
            rate=rate_str,
            volume=volume_str,
            pitch=pitch_str
        )
        await communicate.save(str(output_path))
        return AudioChunk(index=chunk.index, file_path=output_path)

```

**`audio/pydub_merger.py`**

```python
from pathlib import Path
from pydub import AudioSegment
from src.domain.ports import AudioMergerProtocol

class PydubAudioMerger(AudioMergerProtocol):
    def merge(self, audio_files: list[Path], output_path: Path) -> Path:
        combined = AudioSegment.empty()
        for file in sorted(audio_files):
            sound = AudioSegment.from_file(file)
            combined += sound
        combined.export(output_path, format=output_path.suffix.lstrip("."))
        return output_path

```

#### 3. Application Layer (`src/application/`)

**`use_cases.py`**

```python
from pathlib import Path
from src.domain.value_objects import Language, VoiceSettings, ChunkConfig
from src.domain.services import TextSplittingDomainService
from src.domain.ports import TTSEngineProtocol, AudioMergerProtocol, TextSegmenterProtocol

class ConvertTextToSpeechUseCase:
    def __init__(
        self,
        vi_segmenter: TextSegmenterProtocol,
        en_segmenter: TextSegmenterProtocol,
        tts_engine: TTSEngineProtocol,
        audio_merger: AudioMergerProtocol,
    ):
        self._vi_segmenter = vi_segmenter
        self._en_segmenter = en_segmenter
        self._tts_engine = tts_engine
        self._audio_merger = audio_merger

    async def execute(
        self,
        input_path: Path,
        voice_settings: VoiceSettings,
        chunk_config: ChunkConfig,
        should_merge: bool,
    ) -> None:
        files = [input_path] if input_path.is_file() else list(input_path.glob("*.txt"))
        
        for file in files:
            text = file.read_text(encoding="utf-8")
            lang = Language.VIETNAMESE if "tiếng" in text[:100].lower() else Language.ENGLISH # Logic nhận diện đơn giản
            segmenter = self._vi_segmenter if lang == Language.VIETNAMESE else self._en_segmenter
            
            splitter = TextSplittingDomainService(segmenter, chunk_config)
            chunks = splitter.create_chunks(text, lang)

            generated_files: list[Path] = []
            output_dir = file.parent / f"{file.stem}_audio"
            output_dir.mkdir(exist_ok=True)

            for chunk in chunks:
                out_file = output_dir / f"chunk_{chunk.index:03d}.mp3"
                await self._tts_engine.synthesize(chunk, voice_settings, out_file)
                generated_files.append(out_file)

            if should_merge and generated_files:
                merged_path = file.parent / f"{file.stem}_full.mp3"
                self._audio_merger.merge(generated_files, merged_path)

```

#### 4. Presentation Layer (`src/presentation/cli.py`)

```python
import asyncio
from pathlib import Path
import click

from src.domain.value_objects import VoiceSettings, ChunkConfig, TTSEngineType
from src.application.use_cases import ConvertTextToSpeechUseCase
from src.infrastructure.segmenters.nlp_segmenters import VietnameseSegmenter, EnglishSegmenter
from src.infrastructure.tts_engines.edge_engine import EdgeTTSEngine
from src.infrastructure.audio.pydub_merger import PydubAudioMerger

@click.command()
@click.argument("path", type=click.Path(exists=True, path_type=Path))
@click.option("--tts-engine", type=click.Choice([e.value for e in TTSEngineType]), default=TTSEngineType.EDGE.value)
@click.option("--voice", type=str, default=None, help="Mã voice hệ thống/engine")
@click.option("--rate", type=float, default=1.0, help="Tốc độ nói (VD: 1.0, 1.2)")
@click.option("--volume", type=float, default=1.0, help="Âm lượng (VD: 1.0)")
@click.option("--pitch", type=float, default=0.0, help="Độ cao (Hz)")
@click.option("--merge/--no-merge", default=False, help="Gộp các file audio chunk thành 1 file")
@click.option("--chunk-size", type=int, default=2000, help="Kích thước chunk mặc định")
def cli(path: Path, tts_engine: str, voice: str | None, rate: float, volume: float, pitch: float, merge: bool, chunk_size: int):
    """Ứng dụng chuyển đổi Văn bản thành Giọng nói (Text-to-Speech) CLI."""
    
    # Factory / Dependency Injection container đơn giản
    vi_seg = VietnameseSegmenter()
    en_seg = EnglishSegmenter()
    merger = PydubAudioMerger()
    
    # Chọn TTS Adapter (Open-Closed Principle)
    if tts_engine == TTSEngineType.EDGE:
        engine = EdgeTTSEngine()
    else:
        raise NotImplementedError(f"Engine {tts_engine} đang được phát triển.")

    use_case = ConvertTextToSpeechUseCase(
        vi_segmenter=vi_seg,
        en_segmenter=en_seg,
        tts_engine=engine,
        audio_merger=merger
    )

    voice_settings = VoiceSettings(voice=voice, rate=rate, volume=volume, pitch=pitch)
    
    # Validate chunk size max 2500
    max_size = min(chunk_size + 500, 2500)
    chunk_config = ChunkConfig(min_size=1500, default_size=chunk_size, max_size=max_size)

    asyncio.run(use_case.execute(
        input_path=path,
        voice_settings=voice_settings,
        chunk_config=chunk_config,
        should_merge=merge
    ))
    click.echo("Hoàn thành chuyển đổi!")

if __name__ == "__main__":
    cli()

```
