#### **Domain Layer**

`src/domain/models.py`

```python
from enum import StrEnum
from dataclasses import dataclass
from pathlib import Path

class Language(StrEnum):
    VIETNAMESE = "vi"
    ENGLISH = "en"

class TTSEngineType(StrEnum):
    SYSTEM = "system"
    EDGE = "edge"
    GTTS = "gtts"
    GCP = "gcp"

@dataclass(frozen=True, kw_only=True)
class AudioOptions:
    voice: str | None = None
    rate: float = 1.0
    volume: float = 1.0
    pitch: float = 0.0

@dataclass(frozen=True, kw_only=True)
class ChunkConfig:
    min_size: int = 1500
    default_size: int = 2000
    max_size: int = 2500

    def __post_init__() -> None:
        if not (1500 <= self.min_size <= self.default_size <= self.max_size <= 2500):
            raise ValueError("Kích thước chunk không hợp lệ (min >= 1500, max <= 2500)")

@dataclass(frozen=True)
class TextChunk:
    index: int
    content: str
    language: Language

```

#### **Infrastructure Layer - Text Splitters**

`src/domain/services.py`

```python
from src.domain.models import TextChunk, ChunkConfig, Language

class BaseTextSplitter:
    """Thuật toán gom câu giữ nguyên ngữ cảnh, không vượt ngưỡng max_size."""

    def _split_sentences_to_chunks(self, sentences: list[str], config: ChunkConfig, language: Language) -> list[TextChunk]:
        chunks: list[TextChunk] = []
        current_sentences: list[str] = []
        current_len = 0
        chunk_idx = 0

        for sentence in sentences:
            sentence = sentence.strip()
            if not sentence:
                continue

            sentence_len = len(sentence)

            if sentence_len > config.max_size:
                if current_sentences:
                    chunks.append(TextChunk(chunk_idx, " ".join(current_sentences), language))
                    chunk_idx += 1
                    current_sentences, current_len = [], 0

                for sub in [sentence[i:i + config.default_size] for i in range(0, len(sentence), config.default_size)]:
                    chunks.append(TextChunk(chunk_idx, sub, language))
                    chunk_idx += 1
                continue

            if current_len + sentence_len + 1 > config.max_size:
                chunks.append(TextChunk(chunk_idx, " ".join(current_sentences), language))
                chunk_idx += 1
                current_sentences = [sentence]
                current_len = sentence_len
            else:
                current_sentences.append(sentence)
                current_len += sentence_len + 1

        if current_sentences:
            chunks.append(TextChunk(chunk_idx, " ".join(current_sentences), language))

        return chunks

```

`src/infrastructure/text_splitters.py`

```python
import spacy
from underthesea import sent_tokenize
from src.domain.services import BaseTextSplitter
from src.domain.models import TextChunk, ChunkConfig, Language

class VietnameseTextSplitter(BaseTextSplitter):
    def split(self, text: str, config: ChunkConfig, language: Language = Language.VIETNAMESE) -> list[TextChunk]:
        sentences = sent_tokenize(text)
        return self._split_sentences_to_chunks(sentences, config, language)

class EnglishTextSplitter(BaseTextSplitter):
    def __init__(self) -> None:
        self.nlp = spacy.load("en_core_web_sm", disable=["ner", "tagger", "lemmatizer"])

    def split(self, text: str, config: ChunkConfig, language: Language = Language.ENGLISH) -> list[TextChunk]:
        doc = self.nlp(text)
        sentences = [sent.text.strip() for sent in doc.sents]
        return self._split_sentences_to_chunks(sentences, config, language)

```

#### **Infrastructure Layer - TTS Engines & Factory**

`src/infrastructure/tts_engines.py`

```python
import asyncio
from pathlib import Path
import edge_tts
from gtts import gTTS
import pyttsx3

from src.domain.models import TextChunk, AudioOptions, TTSEngineType, Language
from src.domain.interfaces import ITTSEngine

class EdgeTTSEngine(ITTSEngine):
    async def synthesize(self, chunk: TextChunk, options: AudioOptions, output_path: Path) -> Path:
        voice = options.voice or ("vi-VN-HoaiMyNeural" if chunk.language == Language.VIETNAMESE else "en-US-AriaNeural")
        rate_str = f"{int((options.rate - 1) * 100):+d}%"
        volume_str = f"{int((options.volume - 1) * 100):+d}%"
        pitch_str = f"{int(options.pitch):+d}Hz"

        communicate = edge_tts.Communicate(
            text=chunk.content,
            voice=voice,
            rate=rate_str,
            volume=volume_str,
            pitch=pitch_str
        )
        await communicate.save(str(output_path))
        return output_path

class GTTSEngine(ITTSEngine):
    async def synthesize(self, chunk: TextChunk, options: AudioOptions, output_path: Path) -> Path:
        def _generate() -> None:
            tts = gTTS(text=chunk.content, lang=chunk.language.value)
            tts.save(str(output_path))
        
        await asyncio.to_thread(_generate)
        return output_path

class TTSEngineFactory:
    @staticmethod
    def create(engine_type: TTSEngineType) -> ITTSEngine:
        match engine_type:
            case TTSEngineType.EDGE:
                return EdgeTTSEngine()
            case TTSEngineType.GTTS:
                return GTTSEngine()
            case _:
                raise NotImplementedError(f"Engine '{engine_type}' chưa được triển khai.")

```

`src/application/use_cases.py`

```python
import asyncio
from pathlib import Path
from pydub import AudioSegment

from src.application.dtos import ConvertRequestDTO
from src.domain.models import ChunkConfig, AudioOptions, Language, TTSEngineType
from src.domain.interfaces import ITextSplitter
from src.infrastructure.tts_engines import TTSEngineFactory

class ConvertTextToSpeechUseCase:
    def __init__(self, vi_splitter: ITextSplitter, en_splitter: ITextSplitter) -> None:
        self.vi_splitter = vi_splitter
        self.en_splitter = en_splitter

    async def execute(self, dto: ConvertRequestDTO) -> None:
        lang = Language(dto.language)
        splitter = self.vi_splitter if lang == Language.VIETNAMESE else self.en_splitter
        engine = TTSEngineFactory.create(TTSEngineType(dto.engine))
        config = ChunkConfig(default_size=dto.chunk_size)
        options = AudioOptions(
            voice=dto.voice,
            rate=dto.rate,
            volume=dto.volume,
            pitch=dto.pitch
        )

        files = [dto.input_path] if dto.input_path.is_file() else list(dto.input_path.glob("*.txt"))

        for file in files:
            text = file.read_text(encoding="utf-8")
            chunks = splitter.split(text, config, lang)

            output_dir = file.parent / f"{file.stem}_audio"
            output_dir.mkdir(exist_ok=True)

            audio_files: list[Path] = [output_dir / f"chunk_{c.index:04d}.mp3" for c in chunks]

            # Xử lý song song các chunk bằng TaskGroup của Python 3.11+ / 3.13
            async with asyncio.TaskGroup() as tg:
                for chunk, out_file in zip(chunks, audio_files):
                    tg.create_task(engine.synthesize(chunk, options, out_file))

            if dto.merge and audio_files:
                await asyncio.to_thread(self._merge_audio, audio_files, file.parent / f"{file.stem}_merged.mp3")

    @staticmethod
    def _merge_audio(files: list[Path], output_path: Path) -> None:
        combined = AudioSegment.empty()
        for f in files:
            combined += AudioSegment.from_file(f)
        combined.export(output_path, format="mp3")

```

#### **Presentation Layer - CLI**

`src/presentation/cli.py`

```python
import asyncio
import click
from pathlib import Path

from src.application.dtos import ConvertRequestDTO
from src.application.use_cases import ConvertTextToSpeechUseCase
from src.infrastructure.text_splitters import VietnameseTextSplitter, EnglishTextSplitter

@click.command()
@click.option("--input", "-i", "input_path", required=True, type=click.Path(exists=True, path_type=Path), help="File hoặc Folder chứa file text")
@click.option("--lang", "-l", type=click.Choice(["vi", "en"]), default="vi", help="Ngôn ngữ văn bản")
@click.option("--tts-engine", type=click.Choice(["system", "edge", "gtts", "gcp"]), default="edge", help="TTS Engine sử dụng")
@click.option("--voice", type=str, default=None, help="Tên giọng đọc")
@click.option("--rate", type=float, default=1.0, help="Tốc độ đọc")
@click.option("--volume", type=float, default=1.0, help="Âm lượng")
@click.option("--pitch", type=float, default=0.0, help="Độ cao giọng")
@click.option("--merge", is_flag=True, help="Trộn tất cả chunk thành 1 file MP3 duy nhất")
@click.option("--chunk-size", type=click.IntRange(1500, 2500), default=2000, help="Độ dài tối đa chunk (1500 - 2500)")
def main(
    input_path: Path,
    lang: str,
    tts_engine: str,
    voice: str | None,
    rate: float,
    volume: float,
    pitch: float,
    merge: bool,
    chunk_size: int
) -> None:
    dto = ConvertRequestDTO(
        input_path=input_path,
        language=lang,
        engine=tts_engine,
        voice=voice,
        rate=rate,
        volume=volume,
        pitch=pitch,
        merge=merge,
        chunk_size=chunk_size
    )

    use_case = ConvertTextToSpeechUseCase(
        vi_splitter=VietnameseTextSplitter(),
        en_splitter=EnglishTextSplitter()
    )

    asyncio.run(use_case.execute(dto))
    click.echo("✓ Hoàn tất chuyển đổi văn bản sang âm thanh thành công!")

if __name__ == "__main__":
    main()

```
