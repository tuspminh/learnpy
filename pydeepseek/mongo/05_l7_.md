def listen_to_changes(collection_name):
    """
    Lắng nghe tất cả thay đổi trên một collection
    """
    collection = db[collection_name]
    
    print(f"👂 Bắt đầu lắng nghe collection '{collection_name}'...")
    print("   (Nhấn Ctrl+C để dừng)\n")
    
    # Tạo Change Stream
    with collection.watch() as stream:
        for change in stream:
            print(f"📢 [{datetime.now().strftime('%H:%M:%S')}] PHÁT HIỆN THAY ĐỔI:")
            print(f"   - Operation: {change['operationType']}")
            print(f"   - Document ID: {change.get('documentKey', {}).get('_id')}")
            
            # In chi tiết theo từng loại operation
            if change['operationType'] == 'insert':
                print(f"   - Dữ liệu mới: {json.dumps(change['fullDocument'], default=json_util.default, indent=2)}")
            elif change['operationType'] == 'update':
                print(f"   - Fields thay đổi: {list(change['updateDescription']['updatedFields'].keys())}")
                print(f"   - Fields bị xóa: {change['updateDescription']['removedFields']}")
            elif change['operationType'] == 'delete':
                print(f"   - Document đã bị xóa (không thể lấy dữ liệu)")
            
            print("-" * 50)

def listen_to_specific_changes():
    """
    Lắng nghe và lọc chỉ các thay đổi mong muốn

    # Pipeline lọc: Chỉ lắng nghe khi status được update thành 'shipped'

        {
            "$match": {
                "$and": [
                    {"operationType": "update"},
                    {"updateDescription.updatedFields.status": "shipped"}
                ]
            }
        },
        {
            "$project": {
                "order_id": "$documentKey._id",
                "status": "$updateDescription.updatedFields.status",
                "order_details": "$fullDocument"
            }
        }
    ]
    
    print("📦 Lắng nghe đơn hàng được chuyển sang trạng thái 'shipped'...")

        """Xử lý khi đơn hàng được giao"""
        order = change['fullDocument']
        print(f"\n🚚 {datetime.now().strftime('%H:%M:%S')} - ĐƠN HÀNG ĐÃ GIAO!")
        print(f"   - Order ID: {order['_id']}")
        print(f"   - Khách hàng: {order.get('customer_name', 'N/A')}")

            content=f"Đơn hàng #{order['_id']} của bạn đã được giao thành công. Cảm ơn bạn!"

        print(f"   - Order ID: {order['_id']}")
        print(f"   - Lý do: {order.get('cancel_reason', 'Không rõ lý do')}")

        """Bắt đầu lắng nghe tất cả sự kiện"""
        orders_collection = self.db['orders']

                "$match": {
                    "operationType": {"$in": ["insert", "update"]}

                            if new_status == 'shipped':
                                self.on_order_shipped(change)
                            elif new_status == 'cancelled':
                                self.on_order_cancelled(change)

### BƯỚC 6: CHANGE STREAMS NÂNG CAO - RESUME TOKEN

Một tính năng cực kỳ mạnh mẽ: Change Streams có thể **resume** sau khi mất kết nối!

```python
def listen_with_resume_token(collection_name, resume_token=None):
    """
    Lắng nghe với khả năng resume sau khi mất kết nối
    """
    collection = db[collection_name]
    
    print("🔌 Khởi động Change Stream với resume token...")
    
    # Nếu có resume token, tiếp tục từ đó
    if resume_token:
        print(f"   Tiếp tục từ token: {resume_token[:20]}...")
        with collection.watch(resume_after=resume_token) as stream:
            for change in stream:
                print(f"📢 {change['operationType']} - {change.get('documentKey', {}).get('_id')}")
                # Lưu token mới nhất để dùng sau
                resume_token = stream.resume_token
    else:
        # Lần đầu chạy
        with collection.watch() as stream:
            for change in stream:
                print(f"📢 {change['operationType']} - {change.get('documentKey', {}).get('_id')}")
                resume_token = stream.resume_token
    
    return resume_token
```
