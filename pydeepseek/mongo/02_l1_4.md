#### 1.1 So sánh (Comparison Operators)

```python
# Lọc sách theo giá
print("📱 Điện thoại giá từ 15-25 triệu:")
products = collection.find({
    "price": {"$gte": 15000000, "$lte": 25000000}
})
for p in products:

print("\n💪 Điện thoại RAM >= 12GB:")
```

#### 1.2 Logic (Logical Operators)

```python
# $or: Hoặc Samsung, hoặc giá < 20 triệu
print("\n🔄 Samsung hoặc giá dưới 20tr:")
products = collection.find({
    "$or": [
        {"brand": "Samsung"},
        {"price": {"$lt": 20000000}}
    ]
})
for p in products:
    print(f"  - {p['name']}: {p['brand']} - {p['price']:,}đ")

# $and: Vừa Apple vừa có stock > 10
print("\n🍎 Apple còn hàng nhiều (>10):")
products = collection.find({
    "$and": [
        {"brand": "Apple"},
        {"stock": {"$gt": 10}}
    ]
})
for p in products:
    print(f"  - {p['name']}: còn {p['stock']} máy")
```

### CẤP ĐỘ 2: AGGREGATION - "Siêu năng lực" của MongoDB

Aggregation giống như SQL `GROUP BY` nhưng mạnh hơn nhiều. Đây là thứ phân biệt người mới và pro!

```python
# 2.1: Group theo hãng, tính trung bình giá và tổng stock
print("\n📊 Thống kê theo hãng:")
pipeline = [
    {
        "$group": {
            "_id": "$brand",  # Group theo brand
            "avg_price": {"$avg": "$price"},
            "min_price": {"$min": "$price"},
            "max_price": {"$max": "$price"},
            "total_stock": {"$sum": "$stock"},
            "count": {"$sum": 1}
        }
    },
    {"$sort": {"avg_price": -1}}  # Sắp xếp giá TB giảm dần
]

for result in collection.aggregate(pipeline):
    print(f"\n{result['_id']}:")
    print(f"  - Số mẫu: {result['count']}")
    print(f"  - Giá TB: {result['avg_price']:,.0f}đ")
    print(f"  - Giá thấp nhất: {result['min_price']:,}đ")
    print(f"  - Tổng tồn kho: {result['total_stock']}")
```

```python
# 2.2: Thống kê số điện thoại theo phân khúc giá (khó hơn một chút)
print("\n💵 Thống kê theo phân khúc giá:")
pipeline = [
    {
        "$bucket": {  # Chia vào các "bucket" (nhóm)
            "groupBy": "$price",
            "boundaries": [0, 10000000, 20000000, 30000000, 40000000, 100000000],
            "default": "Siêu cao cấp",
            "output": {
                "count": {"$sum": 1},
                "models": {"$push": "$name"},
                "avg_rating": {"$avg": "$rating"}
            }
        }
    }
]

for bucket in collection.aggregate(pipeline):
    print(f"\n{bucket['_id']}:")
    print(f"  - Số lượng: {bucket['count']}")
    print(f"  - Rating TB: {bucket['avg_rating']:.1f}")
    print(f"  - Danh sách: {', '.join(bucket['models'][:2])}...")
```

```python
# 2.3: Sử dụng $project - chỉnh sửa dữ liệu khi query
print("\n✨ Format lại dữ liệu khi xuất:")
products = collection.aggregate([
    {
        "$project": {
            "name": 1,
            "price": 1,
            "price_vnd": {"$concat": [{"$toString": "$price"}, " VND"]},
            "is_expensive": {"$cond": {"if": {"$gt": ["$price", 25000000]}, "then": "Đắt", "else": "Rẻ"}},
            "num_colors": {"$size": "$colors"}
        }
    },
    {"$limit": 3}
])
for p in products:
    print(f"  - {p['name']}: {p['price_vnd']} - {p['is_expensive']} - {p['num_colors']} màu")
```

### CẤP ĐỘ 3: UPDATE NÂNG CAO - Cập nhật thông minh

```python
# 3.1: $inc - Tăng/giảm số (rất hay dùng khi bán hàng)
print("🛒 Bán 1 iPhone 15 Pro...")
collection.update_one(
    {"name": "iPhone 15 Pro"},
    {"$inc": {"stock": -1}}  # Giảm stock đi 1
)

# 3.2: $mul - Nhân giá trị (tăng giá 10%)
print("\n📈 Tăng giá tất cả Samsung lên 10%:")
result = collection.update_many(
    {"brand": "Samsung"},
    {"$mul": {"price": 1.1}}  # Nhân giá với 1.1
)
print(f"Đã cập nhật {result.modified_count} sản phẩm")

# 3.3: $push và $pull - Thêm/xóa phần tử trong array
print("\n➕ Thêm màu 'Vàng' vào iPhone 15:")
collection.update_one(
    {"name": "iPhone 15"},
    {"$push": {"colors": "Vàng"}}
)

print("➖ Xóa màu 'Hồng' khỏi iPhone 15:")
collection.update_one(
    {"name": "iPhone 15"},
    {"$pull": {"colors": "Hồng"}}
)

# 3.4: $addToSet - Thêm vào array nhưng không trùng lặp
print("\n🔄 Thêm màu 'Đen' nếu chưa có:")
collection.update_one(
    {"name": "iPhone 15"},
    {"$addToSet": {"colors": "Đen"}}  # Đã có Đen nên không thêm gì
)

# 3.5: Upsert - Cập nhật nếu có, thêm mới nếu không
print("\n📝 Upsert: Cập nhật hoặc tạo mới sản phẩm:")
result = collection.update_one(
    {"name": "Xiaomi 13 Ultra"},  # Tìm tên này
    {"$set": {"brand": "Xiaomi", "price": 18990000, "stock": 20}},
    upsert=True  # Nếu không thấy thì tự tạo mới
)
print(f"Đã {'cập nhật' if result.matched_count else 'thêm mới'} sản phẩm")
```

### CẤP ĐỘ 4: PERFORMANCE - Index & Query Optimization

**Đây là kiến thức vàng** giúp bạn query nhanh gấp 1000 lần khi database có triệu bản ghi.

```python
# 4.1: Tạo Index (giống mục lục trong sách)
print("\n📇 Tạo Index cho price và brand:")
collection.create_index([("price", -1)])  # -1: giảm dần, 1: tăng dần
collection.create_index([("brand", 1), ("price", -1)])  # Composite index

# Xem tất cả index đã tạo
print("Các index hiện có:")
for idx in collection.list_indexes():
    print(f"  - {idx['name']}: {idx['key']}")

# 4.2: Explain - Xem MongoDB thực hiện query như thế nào
print("\n🔍 Phân tích query plan:")
explain_result = collection.find({"brand": "Apple", "price": {"$gt": 20000000}}).explain()
print(f"  - Phương thức: {explain_result.get('queryPlanner', {}).get('winningPlan', {}).get('stage')}")
print(f"  - Tổng số document xem xét: {explain_result.get('executionStats', {}).get('totalDocsExamined')}")
```

### 🎯 KẾT LUẬN - Lộ trình tiếp theo của bạn

| Cấp độ | Nội dung | Status |
|--------|----------|--------|
| ✅ Level 1 | Các toán tử cơ bản ($gt, $in, $or, $size) | Đã học |
| ✅ Level 2 | Aggregation ($group, $bucket, $project) | Đã học |
| ✅ Level 3 | Update nâng cao ($inc, $push, $pull, upsert) | Đã học |
| ✅ Level 4 | Index & Explain | Đã học |
| ⏳ Level 5 | **Relationship** (Embedded vs Referenced) | *Chưa học* |
| ⏳ Level 6 | **Transactions** (Quan trọng cho tài chính) | *Chưa học* |
| ⏳ Level 7 | **Change Streams** (Realtime) | *Chưa học* |
