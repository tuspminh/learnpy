Dưới đây là chi tiết mã nguồn triển khai cho 2 thành phần cốt lõi: **State Management** (Infrastructure) và **TTS Orchestrator** (Application Use Case) tuân thủ kiến trúc Clean Architecture & DDD.

### 1. State Management (Persistence & State Transfer)

Sử dụng JSON-based repository để lưu trạng thái xử lý từng chunk, hỗ trợ **Resume** khi bị gián đoạn và **Retry** các chunk thất bại.

```python
# infrastructure/persistence/state_repository.py
import json
from pathlib import Path
from typing import List, Optional
from domain.entities import TextChunk, ChunkStatus

class StateRepository:
    def __init__(self, state_file: Path = Path(".tts_state.json")):
        self.state_file = state_file

    def _load_all(self) -> dict:
        if not self.state_file.exists():
            return {}
        try:
            with open(self.state_file, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            return {}

    def _save_all(self, data: dict) -> None:
        with open(self.state_file, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def save_job_state(self, job_id: str, chunks: List[TextChunk]) -> None:
        data = self._load_all()
        data[job_id] = [
            {
                "chunk_id": chunk.chunk_id,
                "file_path": chunk.file_path,
                "sequence_index": chunk.sequence_index,
                "content": chunk.content,
                "status": chunk.status.value,
                "output_audio_path": chunk.output_audio_path,
                "error_message": chunk.error_message,
                "retry_count": chunk.retry_count
            }
            for chunk in chunks
        ]
        self._save_all(data)

    def update_chunk_status(self, job_id: str, chunk: TextChunk) -> None:
        data = self._load_all()
        if job_id not in data:
            return

        for idx, item in enumerate(data[job_id]):
            if item["chunk_id"] == chunk.chunk_id:
                data[job_id][idx].update({
                    "status": chunk.status.value,
                    "output_audio_path": chunk.output_audio_path,
                    "error_message": chunk.error_message,
                    "retry_count": chunk.retry_count
                })
                break
        self._save_all(data)

    def load_job_chunks(self, job_id: str) -> Optional[List[TextChunk]]:
        data = self._load_all()
        if job_id not in data:
            return None

        chunks = []
        for item in data[job_id]:
            chunks.append(
                TextChunk(
                    chunk_id=item["chunk_id"],
                    file_path=item["file_path"],
                    sequence_index=item["sequence_index"],
                    content=item["content"],
                    status=ChunkStatus(item["status"]),
                    output_audio_path=item.get("output_audio_path"),
                    error_message=item.get("error_message"),
                    retry_count=item.get("retry_count", 0)
                )
            )
        return chunks

```

### 2. TTS Orchestrator (Application Core Workflow)

Orchestrator điều phối toàn bộ vòng đời:

* Khởi tạo hoặc khôi phục State (`--resume`).
* Chạy tổng hợp giọng nói cho từng chunk với cơ chế Retry tại chỗ (`auto_retry`).
* Chỉ thực thi các chunk chưa hoàn thành hoặc bị lỗi (`--retry-failed`).
* Gọi `AudioMerger` nối audio sau khi tất cả chunk trong từng file kết thúc thành công.

```python
# application/tts_orchestrator.py
import asyncio
import hashlib
from pathlib import Path
from typing import List, Dict
from domain.entities import TextChunk, ChunkStatus
from domain.value_objects import VoiceConfig
from domain.services import BaseTTSProvider
from infrastructure.persistence.state_repository import StateRepository
from infrastructure.nlp.text_splitter import SmartTextSplitter
from infrastructure.audio.merger import AudioMerger

class TTSOrchestrator:
    def __init__(
        self,
        provider: BaseTTSProvider,
        state_repo: StateRepository,
        text_splitter: SmartTextSplitter,
        audio_merger: AudioMerger,
        temp_dir: Path = Path(".tts_temp")
    ):
        self.provider = provider
        self.state_repo = state_repo
        self.text_splitter = text_splitter
        self.audio_merger = audio_merger
        self.temp_dir = temp_dir
        self.temp_dir.mkdir(parents=True, exist_ok=True)

    def _generate_job_id(self, files: List[Path]) -> str:
        raw_str = "|".join(sorted([f.resolve().as_posix() for f in files]))
        return hashlib.md5(raw_str.encode("utf-8")).hexdigest()

    async def execute(
        self,
        file_paths: List[Path],
        config: VoiceConfig,
        output_dir: Path,
        max_retries: int = 3,
        resume: bool = True,
        retry_failed_only: bool = False
    ) -> None:
        job_id = self._generate_job_id(file_paths)
        output_dir.mkdir(parents=True, exist_ok=True)

        # 1. State Transfer & Recovery Setup
        existing_chunks = self.state_repo.load_job_chunks(job_id) if resume else None

        if existing_chunks:
            chunks = existing_chunks
            print(f"[State] Resumed job {job_id} with {len(chunks)} chunks.")
        else:
            chunks = self._prepare_chunks(file_paths, config.language.value)
            self.state_repo.save_job_state(job_id, chunks)
            print(f"[State] Created new job {job_id} with {len(chunks)} chunks.")

        # 2. Group chunks by target text file
        grouped_chunks: Dict[str, List[TextChunk]] = {}
        for chunk in chunks:
            grouped_chunks.setdefault(chunk.file_path, []).append(chunk)

        # 3. Process Chunk Conversion Loop
        for file_path, file_chunks in grouped_chunks.items():
            print(f"\n---> Processing file: {file_path}")
            file_success = True

            for chunk in file_chunks:
                # Filter execution based on flags
                if chunk.status == ChunkStatus.COMPLETED and not retry_failed_only:
                    continue
                if retry_failed_only and chunk.status != ChunkStatus.FAILED:
                    continue

                chunk_success = await self._process_single_chunk_with_retry(
                    job_id=job_id,
                    chunk=chunk,
                    config=config,
                    max_retries=max_retries
                )

                if not chunk_success:
                    file_success = False

            # 4. Merge Audio for current file if all chunks succeeded
            if file_success and all(c.status == ChunkStatus.COMPLETED for c in file_chunks):
                merged_output_name = Path(file_path).stem + ".mp3"
                final_output_path = output_dir / merged_output_name
                
                ordered_audio_paths = [
                    Path(c.output_audio_path)
                    for c in sorted(file_chunks, key=lambda x: x.sequence_index)
                ]

                print(f"[Merger] Joining {len(ordered_audio_paths)} chunks -> {final_output_path}")
                self.audio_merger.merge(ordered_audio_paths, final_output_path)
            else:
                print(f"[Warning] File {file_path} skipped audio merging due to failed chunks.")

    def _prepare_chunks(self, file_paths: List[Path], lang: str) -> List[TextChunk]:
        all_chunks = []
        for file_path in file_paths:
            content = file_path.read_text(encoding="utf-8")
            raw_segments = self.text_splitter.split(content, lang)

            for idx, text_segment in enumerate(raw_segments):
                chunk_id = hashlib.sha256(f"{file_path}_{idx}_{text_segment[:20]}".encode()).hexdigest()[:12]
                all_chunks.append(
                    TextChunk(
                        chunk_id=chunk_id,
                        file_path=str(file_path.resolve()),
                        sequence_index=idx,
                        content=text_segment,
                        status=ChunkStatus.PENDING
                    )
                )
        return all_chunks

    async def _process_single_chunk_with_retry(
        self,
        job_id: str,
        chunk: TextChunk,
        config: VoiceConfig,
        max_retries: int
    ) -> bool:
        chunk.status = ChunkStatus.PROCESSING
        self.state_repo.update_chunk_status(job_id, chunk)

        out_path = self.temp_dir / f"{chunk.chunk_id}.mp3"

        while chunk.retry_count <= max_retries:
            try:
                print(f"  [TTS] Converting chunk {chunk.sequence_index + 1} (Attempt {chunk.retry_count + 1})...")
                
                success = await self.provider.synthesize(
                    text=chunk.content,
                    config=config,
                    output_path=str(out_path)
                )

                if success and out_path.exists():
                    chunk.status = ChunkStatus.COMPLETED
                    chunk.output_audio_path = str(out_path.resolve())
                    chunk.error_message = None
                    self.state_repo.update_chunk_status(job_id, chunk)
                    return True

            except Exception as e:
                chunk.error_message = str(e)
                print(f"  [Error] Chunk {chunk.chunk_id} failed: {e}")

            chunk.retry_count += 1
            if chunk.retry_count <= max_retries:
                await asyncio.sleep(2 ** chunk.retry_count)  # Exponential backoff

        chunk.status = ChunkStatus.FAILED
        self.state_repo.update_chunk_status(job_id, chunk)
        return False
```
