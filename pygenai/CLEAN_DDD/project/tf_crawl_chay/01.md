```python
from domain.value_objects import Slug, ChapterNo, Url

class Chapter:
    """Entity: Chapter"""

    title: str

    url: Optional[Url] = None

@dataclass
class Novel:
    """Aggregate Root: Novel"""
    slug: Slug
    title: str
    author: str
    description: str
    cover_url: Url
```

#### `domain/ports.py`

```python
from abc import ABC, abstractmethod
from typing import List, Optional
from domain.entities import Novel, Chapter, Listing
from domain.value_objects import Slug, Url

class HttpFetcherPort(ABC):
    @abstractmethod
    def get(self, url: str) -> str:
        pass

class ParserPort(ABC):
    @abstractmethod
    def parse_listing_novel_slugs(self, html: str) -> List[Slug]:
        pass

    @abstractmethod
    def parse_listing_max_page(self, html: str) -> int:
        pass

    @abstractmethod
    def parse_novel_info(self, html: str, slug: Slug) -> Novel:
        pass

    @abstractmethod
    def parse_novel_chapter_links(self, html: str) -> List[Chapter]:
        pass

    @abstractmethod
    def parse_novel_max_chapter_page(self, html: str) -> int:
        pass

    @abstractmethod
    def parse_chapter_content(self, html: str) -> str:
        pass

class NovelRepositoryPort(ABC):
    @abstractmethod
    def save_novel(self, novel: Novel) -> None:
        pass

    @abstractmethod
    def save_chapter(self, novel_slug: Slug, chapter: Chapter) -> None:
        pass

```

### 2. Tầng Application (`application/`)

#### `application/use_cases.py`

```python
from domain.ports import HttpFetcherPort, ParserPort, NovelRepositoryPort
from domain.value_objects import Slug
from domain.entities import Novel

class CrawlNovelUseCase:
    """Use Case: Crawl toàn bộ thông tin truyện + danh sách chương + nội dung từng chương"""
    def __init__(
        self,
        base_url: str,
        fetcher: HttpFetcherPort,
        parser: ParserPort,
        repository: NovelRepositoryPort
    ):
        self.base_url = base_url.rstrip("/")
        self.fetcher = fetcher
        self.parser = parser
        self.repository = repository

    def execute(self, novel_slug_str: str) -> Novel:
        slug = Slug(novel_slug_str)
        novel_url = f"{self.base_url}/{slug.value}/"
        
        # 1. Crawl trang 1 novel (Thông tin + danh sách chapter trang 1)
        html_page_1 = self.fetcher.get(novel_url)
        novel = self.parser.parse_novel_info(html_page_1, slug)
        chapters_page_1 = self.parser.parse_novel_chapter_links(html_page_1)
        novel.chapters.extend(chapters_page_1)

        # 2. Xử lý phân trang danh sách chapter (site/novel-slug/trang-2/)
        max_pages = self.parser.parse_novel_max_chapter_page(html_page_1)
        for page in range(2, max_pages + 1):
            page_url = f"{self.base_url}/{slug.value}/trang-{page}/"
            html_page = self.fetcher.get(page_url)
            chapters = self.parser.parse_novel_chapter_links(html_page)
            novel.chapters.extend(chapters)

        self.repository.save_novel(novel)

        # 3. Crawl nội dung từng chapter (site/novel-slug/chuong-no/)
        for chapter in novel.chapters:
            if chapter.url:
                chapter_html = self.fetcher.get(chapter.url.value)
                chapter.content = self.parser.parse_chapter_content(chapter_html)
                self.repository.save_chapter(slug, chapter)

        return novel


class CrawlListingUseCase:
    """Use Case: Crawl toàn bộ danh sách truyện từ 1 Listing"""
    def __init__(
        self,
        base_url: str,
        fetcher: HttpFetcherPort,
        parser: ParserPort,
        crawl_novel_uc: CrawlNovelUseCase
    ):
        self.base_url = base_url.rstrip("/")
        self.fetcher = fetcher
        self.parser = parser
        self.crawl_novel_uc = crawl_novel_uc

    def execute(self, listing_slug_str: str):
        listing_url = f"{self.base_url}/{listing_slug_str}/"
        html_page_1 = self.fetcher.get(listing_url)

        max_pages = self.parser.parse_listing_max_page(html_page_1)

        for page in range(1, max_pages + 1):
            page_url = f"{listing_url}" if page == 1 else f"{self.base_url}/{listing_slug_str}/trang-{page}/"
            html = self.fetcher.get(page_url)
            novel_slugs = self.parser.parse_listing_novel_slugs(html)
            
            for slug in novel_slugs:
                self.crawl_novel_uc.execute(slug.value)

```

#### `infrastructure/http_fetcher.py`

```python
import urllib.request
from domain.ports import HttpFetcherPort

class RequestsFetcherAdapter(HttpFetcherPort):
    def __init__(self, headers: dict = None):
        self.headers = headers or {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"}

    def get(self, url: str) -> str:
        req = urllib.request.Request(url, headers=self.headers)
        with urllib.request.urlopen(req) as response:
            return response.read().decode('utf-8')

```

#### `infrastructure/parsers.py`

```python
import re
from bs4 import BeautifulSoup
from typing import List
from domain.ports import ParserPort
from domain.entities import Novel, Chapter
from domain.value_objects import Slug, ChapterNo, Url

class BeautifulSoupTruyenFullParserAdapter(ParserPort):
    def parse_listing_novel_slugs(self, html: str) -> List[Slug]:
        soup = BeautifulSoup(html, 'html.parser')
        slugs = []
        for elem in soup.select(".list-truyen .truyen-title a"):
            href = elem.get('href', '')
            slug_match = re.search(r'truyenfull\.live/([^/]+)/', href)
            if slug_match:
                slugs.append(Slug(slug_match.group(1)))
        return slugs

    def parse_listing_max_page(self, html: str) -> int:
        soup = BeautifulSoup(html, 'html.parser')
        pagination = soup.select(".pagination a")
        pages = [1]
        for a in pagination:
            href = a.get('href', '')
            match = re.search(r'trang-(\d+)', href)
            if match:
                pages.append(int(match.group(1)))
        return max(pages)

    def parse_novel_info(self, html: str, slug: Slug) -> Novel:
        soup = BeautifulSoup(html, 'html.parser')
        title = soup.select_one("h3.title").get_text(strip=True) if soup.select_one("h3.title") else ""
        author = soup.select_one("a[itemprop='author']").get_text(strip=True) if soup.select_one("a[itemprop='author']") else "Unknown"
        desc = soup.select_one(".desc-text").get_text(strip=True) if soup.select_one(".desc-text") else ""
        cover_elem = soup.select_one(".book img")
        cover_url = Url(cover_elem.get('src') if cover_elem else "")

        return Novel(slug=slug, title=title, author=author, description=desc, cover_url=cover_url)

    def parse_novel_chapter_links(self, html: str) -> List[Chapter]:
        soup = BeautifulSoup(html, 'html.parser')
        chapters = []
        for elem in soup.select("ul.list-chapter li a"):
            href = elem.get('href', '')
            title = elem.get_text(strip=True)
            match = re.search(r'chuong-(\d+)', href)
            ch_num = int(match.group(1)) if match else 0
            chapters.append(Chapter(number=ChapterNo(ch_num), title=title, url=Url(href)))
        return chapters

    def parse_novel_max_chapter_page(self, html: str) -> int:
        return self.parse_listing_max_page(html)

    def parse_chapter_content(self, html: str) -> str:
        soup = BeautifulSoup(html, 'html.parser')
        content_elem = soup.select_one("#chapter-c")
        if content_elem:
            return content_elem.decode_contents()
        return ""

```

#### `infrastructure/storage.py`

```python
import json
import os
from domain.ports import NovelRepositoryPort
from domain.entities import Novel, Chapter
from domain.value_objects import Slug

class JsonFileNovelRepositoryAdapter(NovelRepositoryPort):
    def __init__(self, output_dir: str = "data"):
        self.output_dir = output_dir
        os.makedirs(self.output_dir, exist_ok=True)

    def save_novel(self, novel: Novel) -> None:
        novel_dir = os.path.join(self.output_dir, novel.slug.value)
        os.makedirs(novel_dir, exist_ok=True)
        
        meta = {
            "slug": novel.slug.value,
            "title": novel.title,
            "author": novel.author,
            "description": novel.description,
            "cover_url": novel.cover_url.value,
            "total_chapters": len(novel.chapters)
        }
        with open(os.path.join(novel_dir, "metadata.json"), "w", encoding="utf-8") as f:
            json.dump(meta, f, ensure_ascii=False, indent=2)

    def save_chapter(self, novel_slug: Slug, chapter: Chapter) -> None:
        chapters_dir = os.path.join(self.output_dir, novel_slug.value, "chapters")
        os.makedirs(chapters_dir, exist_ok=True)
        
        file_name = f"chuong_{chapter.number.value}.json"
        data = {
            "number": chapter.number.value,
            "title": chapter.title,
            "content": chapter.content
        }
        with open(os.path.join(chapters_dir, file_name), "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

```

### 4. Tầng Presentation / Entrypoint (`main.py`)

```python
from infrastructure.http_fetcher import RequestsFetcherAdapter
from infrastructure.parsers import BeautifulSoupTruyenFullParserAdapter
from infrastructure.storage import JsonFileNovelRepositoryAdapter
from application.use_cases import CrawlNovelUseCase, CrawlListingUseCase

def main():
    BASE_URL = "https://truyenfull.live"

    # Dependency Injection Setup
    fetcher = RequestsFetcherAdapter()
    parser = BeautifulSoupTruyenFullParserAdapter()
    repository = JsonFileNovelRepositoryAdapter(output_dir="storage_data")

    # Inject Use Cases
    crawl_novel_uc = CrawlNovelUseCase(
        base_url=BASE_URL,
        fetcher=fetcher,
        parser=parser,
        repository=repository
    )
    
    crawl_listing_uc = CrawlListingUseCase(
        base_url=BASE_URL,
        fetcher=fetcher,
        parser=parser,
        crawl_novel_uc=crawl_novel_uc
    )

    # Thực thi: Crawl 1 novel cụ thể
    print("Starting novel crawl...")
    crawl_novel_uc.execute("dau-la-dai-luc")

    # Hoặc Crawl toàn bộ 1 listing (Vd: danh-sach/truyen-hot)
    # crawl_listing_uc.execute("danh-sach/truyen-hot")

if __name__ == "__main__":
    main()
```
